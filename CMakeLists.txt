cmake_minimum_required(VERSION 3.24)
project(laya VERSION 0.1.0 LANGUAGES CXX)
add_library(laya src/decision.cpp)
target_include_directories(laya PUBLIC include)
target_compile_features(laya PUBLIC cxx_std_20)
target_compile_options(laya PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(test-decision tests/decision.cpp)
  target_link_libraries(test-decision PRIVATE laya)
  add_test(NAME decision COMMAND test-decision)
endif()
