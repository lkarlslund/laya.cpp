cmake_minimum_required(VERSION 3.24)
project(laya VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
option(LAYA_CUDA "Build the NVIDIA CUDA inference backend" ON)
set(GGML_CUDA ${LAYA_CUDA} CACHE BOOL "" FORCE)
set(GGML_CUDA_GRAPHS ON CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/ggml)
if(LAYA_CUDA)
  # Extend one CUDA dispatcher without editing the pinned dependency checkout.
  set_property(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ggml/src/ggml-cuda/fattn.cu"
    TARGET_DIRECTORY ggml-cuda APPEND PROPERTY COMPILE_DEFINITIONS
    ggml_cuda_flash_attn_ext=ggml_cuda_flash_attn_ext_upstream)
  # Preserve FP32 accumulators and output for half-input cuBLAS products.
  set(cuda_source "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ggml/src/ggml-cuda/ggml-cuda.cu")
  file(READ "${cuda_source}" cuda_code)
  set(old_policy "prefer_f32_output = cc == GGML_CUDA_CC_VOLTA || GGML_CUDA_CC_IS_RDNA4(cc) || GGML_CUDA_CC_IS_CDNA(cc);")
  string(FIND "${cuda_code}" "${old_policy}" policy_position)
  if(policy_position EQUAL -1)
    message(FATAL_ERROR "Pinned CUDA accumulation policy changed")
  endif()
  string(REPLACE "${old_policy}" "prefer_f32_output = true;" cuda_code "${cuda_code}")
  string(REPLACE "static bool ggml_cuda_compute_forward("
    "bool laya_cuda_custom(ggml_backend_cuda_context&, ggml_tensor*);\nstatic bool ggml_cuda_compute_forward("
    cuda_code "${cuda_code}")
  string(REPLACE "        case GGML_OP_CONCAT:"
    "        case GGML_OP_CUSTOM: return laya_cuda_custom(ctx, dst);\n        case GGML_OP_CONCAT:"
    cuda_code "${cuda_code}")
  # The capability query has different argument names and must not execute kernels.
  string(REPLACE "case GGML_OP_CUSTOM: return laya_cuda_custom(ctx, dst);\n        case GGML_OP_CONCAT:\n            {\n                ggml_type src0_type = op->src[0]->type;"
    "case GGML_OP_CUSTOM: return strcmp(op->name, \"laya.split-f16\") == 0 || strcmp(op->name, \"laya.merge-f16\") == 0 || strcmp(op->name, \"laya.pack-qkv\") == 0 || strcmp(op->name, \"laya.mlp-split\") == 0 || strcmp(op->name, \"laya.mask-local\") == 0 || strcmp(op->name, \"laya.mask-global\") == 0;\n        case GGML_OP_CONCAT:\n            {\n                ggml_type src0_type = op->src[0]->type;"
    cuda_code "${cuda_code}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/laya-ggml-cuda.cu" "${cuda_code}")
  get_target_property(cuda_sources ggml-cuda SOURCES)
  list(TRANSFORM cuda_sources REPLACE "^(.*/)?ggml-cuda[.]cu$" "${CMAKE_CURRENT_BINARY_DIR}/laya-ggml-cuda.cu")
  set_property(TARGET ggml-cuda PROPERTY SOURCES "${cuda_sources}")
  target_sources(ggml-cuda PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/attention.cu" "${CMAKE_CURRENT_SOURCE_DIR}/src/precision.cu")
  target_include_directories(ggml-cuda PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ggml/src/ggml-cuda")
endif()
find_package(nlohmann_json 3.11 REQUIRED)
find_package(ICU REQUIRED COMPONENTS uc i18n)
add_library(laya src/decision.cpp src/precision.cpp src/tokenizer.cpp src/runtime.cpp src/protocol.cpp)
target_include_directories(laya PUBLIC include)
target_compile_features(laya PUBLIC cxx_std_20)
target_compile_options(laya PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(laya PUBLIC ggml nlohmann_json::nlohmann_json ICU::uc ICU::i18n)
if(LAYA_CUDA)
  target_compile_definitions(laya PRIVATE LAYA_CUDA=1)
  target_link_libraries(laya PRIVATE ggml-cuda)
endif()
add_executable(laya-cli src/main.cpp)
target_link_libraries(laya-cli PRIVATE laya)
include(CTest)
if(BUILD_TESTING)
  add_executable(test-decision tests/decision.cpp)
  target_link_libraries(test-decision PRIVATE laya)
  add_test(NAME decision COMMAND test-decision)
  add_executable(test-tokenizer tests/tokenizer.cpp)
  target_link_libraries(test-tokenizer PRIVATE laya)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/models/laya/tokenizer/tokenizer.json")
    add_test(NAME tokenizer COMMAND test-tokenizer
      "${CMAKE_CURRENT_SOURCE_DIR}/models/laya/tokenizer/tokenizer.json"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/tokenizer-cases.json")
  endif()
  foreach(variant multilingual typed-decisions)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/models/laya/${variant}/tokenizer/tokenizer.json")
      if(variant STREQUAL "multilingual")
        set(fixtures tokenizer-multilingual.json)
      else()
        set(fixtures tokenizer-cases.json)
      endif()
      add_test(NAME tokenizer-${variant} COMMAND test-tokenizer
        "${CMAKE_CURRENT_SOURCE_DIR}/models/laya/${variant}/tokenizer/tokenizer.json"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/${fixtures}")
    endif()
  endforeach()
  if(LAYA_CUDA)
    add_executable(test-attention tests/attention.cpp)
    target_link_libraries(test-attention PRIVATE ggml ggml-cuda)
    add_test(NAME attention COMMAND test-attention)
    add_executable(test-fused tests/fused.cpp)
    target_link_libraries(test-fused PRIVATE laya ggml-cuda)
    add_test(NAME fused-operations COMMAND test-fused)
    add_executable(test-matmul tests/matmul.cpp)
    target_link_libraries(test-matmul PRIVATE laya ggml-cuda)
    add_test(NAME compensated-matmul COMMAND test-matmul)
  endif()
endif()
